cmake_minimum_required(VERSION 3.16)
project(stereo_offline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(OpenCV 4 REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system)
set(OV "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/open_vins")
add_executable(stereo_offline stereo_offline.cpp)
target_compile_definitions(stereo_offline PRIVATE ROS_AVAILABLE=0 ENABLE_ARUCO_TAGS=1)
target_include_directories(stereo_offline PRIVATE ${OV}/ov_msckf/src ${OV}/ov_core/src ${OV}/ov_init/src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(stereo_offline PRIVATE Eigen3::Eigen Boost::filesystem Boost::system ${OpenCV_LIBS} "${CMAKE_CURRENT_SOURCE_DIR}/../../build/open_vins/libov_msckf_lib.so")
